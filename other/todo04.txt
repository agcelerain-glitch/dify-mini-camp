  ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

      --- 以下を貼り付け（ファイルから直接コピーしてください） ---

      CREATE TABLE IF NOT EXISTS public.feedback (
        id         UUID        DEFAULT gen_random_uuid() PRIMARY KEY,
        user_id    UUID        NOT NULL REFERENCES public.users(id) ON DELETE CASCADE,
        phase_id   SMALLINT,
        level_id   SMALLINT,
        page_id    SMALLINT,
        rating     SMALLINT    NOT NULL CHECK (rating BETWEEN 1 AND 5),
        comment    TEXT,
        created_at TIMESTAMPTZ DEFAULT NOW()
      );

      ALTER TABLE public.feedback ENABLE ROW LEVEL SECURITY;

      DROP POLICY IF EXISTS "Users can insert own feedback" ON public.feedback;
      DROP POLICY IF EXISTS "Users can view own feedback"   ON public.feedback;

      CREATE POLICY "Users can insert own feedback"
        ON public.feedback FOR INSERT
        WITH CHECK (auth.uid() = user_id);

      CREATE POLICY "Users can view own feedback"
        ON public.feedback FOR SELECT
        USING (auth.uid() = user_id);

      --- ここまで ---

■ よくあるトラブルと対処法
================================================================

  Q: 星ボタンを押しても色が変わらない
  A: フロントの JavaScript エラーを確認。
     ブラウザのデベロッパーツール（F12）→ Console タブを確認。

  Q: 「フィードバックを送信」を押すと 500 エラーになる
  A: Supabase の feedback テーブルが作成されていない可能性。
     STEP A の SQL を Supabase SQL Editor で実行してください。

  Q: feedback テーブルへの保存はできるが「全コース完了！」ボタンが出ない
  A: Supabase の progress テーブルへのレベルクリア保存に失敗している可能性。
     ブラウザのネットワークタブ（F12）でエラーレスポンスを確認。
     Supabase の RLS ポリシーで progress テーブルの UPDATE が許可されているか確認。

  Q: 卒業ページで「楽しかったです」を送ると採点JSON（is_cleared...）が返る
  A: IF/ELSE② が追加されていないか、接続が正しくない。
     STEP C-1〜C-2 を確認し、IF/ELSE② の TRUE 出口が LLM_メンター に
     正しく繋がっているか確認してください。
     その後「公開する」→「公開」を再実行してください。

  Q: Dify メンターが卒業トーンではなく通常の技術回答を返す
  A: STEP C-3 の追記ができていないか、追記後に「公開する」を忘れている。
     Dify で「公開する」→「公開」を再実行してください。

  Q: チャットに送信すると「メンターへの接続に失敗しました」と表示される
  A: Dify の開始ブロックに current_page 変数を追加していないと、
     Dify 側でエラーになる場合がある。
     STEP B を実施し「公開」を押してください。
     ※ current_page は必須オフ（任意）なので変数追加前でも動作する場合がありますが
       念のため STEP B を実施してください。
